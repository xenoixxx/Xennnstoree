    <script>
        const SUPABASE_URL = "https://oshltauwzurnxaqjlwdz.supabase.co";
        const SUPABASE_KEY = "sb_publishable_-WBcVndQ2JVR55yt6QfhFg_k1XB5ANd";
        const supabase = supabase.createClient(SUPABASE_URL, SUPABASE_KEY);

        // Fetch Orders & Products
        async function loadAdminData() {
            // Load Orders
            const { data: orders } = await supabase.from('orders').select('*').order('id', { ascending: false });
            const tbody = document.getElementById('orders-list-body');
            if (orders && orders.length > 0) {
                tbody.innerHTML = orders.map(o => `
                    <tr>
                        <td class="py-3 font-semibold text-white">${o.customer_name}</td>
                        <td class="py-3 text-purple-400">${o.phone_number}</td>
                        <td class="py-3">${o.wilaya}<br><span class="text-zinc-500">${o.address}</span></td>
                        <td class="py-3">#${o.product_id}</td>
                        <td class="py-3 text-zinc-500">${new Date(o.created_at).toLocaleDateString()}</td>
                    </tr>
                `).join('');
            } else {
                tbody.innerHTML = `<tr><td colspan="5" class="py-4 text-center text-zinc-600">No orders yet.</td></tr>`;
            }

            // Load Inventory
            const { data: products } = await supabase.from('products').select('*').order('id', { ascending: false });
            const invContainer = document.getElementById('inventory-list');
            if (products && products.length > 0) {
                invContainer.innerHTML = products.map(p => `
                    <div class="flex justify-between items-center bg-zinc-950 p-3 rounded border border-zinc-800 text-xs">
                        <div>
                            <span class="font-bold text-white">${p.title}</span> — 
                            <span class="text-purple-400">${p.price} DZD</span> (${p.size})
                        </div>
                        <button onclick="toggleAvailability(${p.id}, ${p.is_available})" class="px-2 py-1 rounded text-xs font-bold ${p.is_available ? 'bg-emerald-950 text-emerald-400 border border-emerald-800' : 'bg-red-950 text-red-400 border border-red-800'}">
                            ${p.is_available ? 'In Stock' : 'Sold Out'}
                        </button>
                    </div>
                `).join('');
            } else {
                invContainer.innerHTML = `<p class="text-xs text-zinc-600">No inventory listed.</p>`;
            }
        }

        // Add Product
        async function handleAddProduct(e) {
            e.preventDefault();
            const btn = document.getElementById('p-btn');
            btn.disabled = true;

            const newItem = {
                title: document.getElementById('p-title').value,
                price: parseInt(document.getElementById('p-price').value),
                size: document.getElementById('p-size').value,
                image_url: document.getElementById('p-image').value,
                is_available: true
            };

            const { error } = await supabase.from('products').insert([newItem]);
            if (!error) {
                document.getElementById('add-product-form').reset();
                loadAdminData();
            } else {
                alert("Error adding product. Enable full permissions in Supabase or check your input.");
            }
            btn.disabled = false;
        }

        // Toggle Stock Status
        async function toggleAvailability(id, currentStatus) {
            await supabase.from('products').update({ is_available: !currentStatus }).eq('id', id);
            loadAdminData();
        }

        loadAdminData();
    </script>
