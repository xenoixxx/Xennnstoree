        // Fetch & Render Products
        async function fetchProducts() {
            const grid = document.getElementById('product-grid');
            const { data: products, error } = await supabase.from('products').select('*').order('id', { ascending: false });

            if (error) {
                grid.innerHTML = `<p class="text-red-400 col-span-full text-center">Error loading products.</p>`;
                return;
            }

            if (!products || products.length === 0) {
                grid.innerHTML = `<p class="text-zinc-500 col-span-full text-center py-12">No items listed yet. Check back soon!</p>`;
                return;
            }

            grid.innerHTML = products.map(p => `
                <div class="bg-zinc-900 rounded-xl overflow-hidden y2k-border transition flex flex-col justify-between">
                    <div>
                        <div class="h-64 bg-zinc-950 overflow-hidden relative">
                            <img src="${p.image_url || 'https://via.placeholder.com/400x500?text=No+Photo'}" alt="${p.title}" class="w-full h-full object-cover">
                            ${!p.is_available ? `<span class="absolute top-2 right-2 bg-red-600/90 text-white text-xs font-bold px-2 py-1 rounded">SOLD OUT</span>` : ''}
                        </div>
                        <div class="p-4">
                            <div class="flex justify-between items-start mb-2">
                                <h3 class="font-bold text-white text-lg leading-snug">${p.title}</h3>
                                <span class="bg-zinc-800 text-purple-400 text-xs font-bold px-2 py-1 rounded border border-zinc-700">${p.size}</span>
                            </div>
                            <p class="text-xl font-black text-white">${p.price} DZD</p>
                        </div>
                    </div>
                    <div class="p-4 pt-0">
                        <button 
                            onclick="openCheckout(${p.id}, '${p.title.replace(/'/g, "\\'")}')" 
                            ${!p.is_available ? 'disabled' : ''}
                            class="w-full py-2.5 rounded-lg text-sm font-bold transition ${p.is_available ? 'bg-white text-black hover:bg-purple-500 hover:text-white' : 'bg-zinc-800 text-zinc-500 cursor-not-allowed'}">
                            ${p.is_available ? 'Order Now' : 'Out of Stock'}
                        </button>
                    </div>
                </div>
            `).join('');
        }

        // Modal Handlers
        function openCheckout(productId, title) {
            document.getElementById('selected-product-id').value = productId;
            document.getElementById('modal-product-title').textContent = title;
            document.getElementById('checkout-modal').classList.remove('hidden');
        }

        function closeCheckout() {
            document.getElementById('checkout-modal').classList.add('hidden');
            document.getElementById('order-form').reset();
        }

        // Submit Order to Supabase
        async function handleOrderSubmit(e) {
            e.preventDefault();
            const btn = document.getElementById('submit-btn');
            btn.disabled = true;
            btn.textContent = "Processing...";

            const orderData = {
                product_id: parseInt(document.getElementById('selected-product-id').value),
                customer_name: document.getElementById('cust-name').value,
                phone_number: document.getElementById('cust-phone').value,
                wilaya: document.getElementById('cust-wilaya').value,
                address: document.getElementById('cust-address').value,
                status: 'pending'
            };

            const { error } = await supabase.from('orders').insert([orderData]);

            if (error) {
                alert("Failed to submit order. Please try again.");
                console.error(error);
                btn.disabled = false;
                btn.textContent = "Submit Order (Cash on Delivery)";
            } else {
                alert("Order placed successfully! We will contact you soon.");
                closeCheckout();
                btn.disabled = false;
                btn.textContent = "Submit Order (Cash on Delivery)";
            }
        }

        // Load Products on page open
        fetchProducts();
